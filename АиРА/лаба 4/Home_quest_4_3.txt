from random import random as rnd
from math import hypot, pi

amount, cnt = 10_000_000, 0

for _ in range(amount):

    x, y = rnd(), rnd()

    if hypot(x, y) <= 1.0:

        cnt += 1

Pi = 4 * cnt / amount
